#include <stdio.h>
#define MAX 10

typedef struct {
    int datos[MAX];
    int n;
} Lista;

void iniciar(Lista *l) { l->n = 0; }

void insertar(Lista *l, int pos, int v) {
    if (l->n == MAX) { printf("Lista llena\n"); return; }
    if (pos < 1 || pos > l->n + 1) { printf("Posicion invalida\n"); return; }
    for (int i = l->n; i >= pos; i--) l->datos[i] = l->datos[i - 1];
    l->datos[pos - 1] = v;
    l->n++;
}

int eliminar(Lista *l, int pos, int *v) {
    if (pos < 1 || pos > l->n) return 0;
    *v = l->datos[pos - 1];
    for (int i = pos - 1; i < l->n - 1; i++) l->datos[i] = l->datos[i + 1];
    l->n--;
    return 1;
}

int buscar(Lista *l, int v) {
    for (int i = 0; i < l->n; i++) if (l->datos[i] == v) return i + 1;
    return 0;
}

void mostrar(Lista *l) {
    if (l->n == 0) { printf("Lista vacia\n"); return; }
    for (int i = 0; i < l->n; i++) printf("[%d] ", l->datos[i]);
    printf("\n");
}

int main() {
    Lista l; int op, v, pos;
    iniciar(&l);
    do {
        printf("\n--- LISTA ESTATICA ---\n1.Insertar  2.Eliminar  3.Buscar  4.Mostrar  0.Salir\nOpcion: ");
        scanf("%d", &op);
        switch (op) {
            case 1: printf("Posicion (1-%d): ", l.n + 1); scanf("%d", &pos);
                    printf("Valor: "); scanf("%d", &v); insertar(&l, pos, v); break;
            case 2: printf("Posicion: "); scanf("%d", &pos);
                    if (eliminar(&l, pos, &v)) printf("Eliminado: %d\n", v); else printf("Posicion invalida\n"); break;
            case 3: printf("Valor: "); scanf("%d", &v); pos = buscar(&l, v);
                    if (pos) printf("Encontrado en posicion %d\n", pos); else printf("No encontrado\n"); break;
            case 4: mostrar(&l); break;
        }
    } while (op != 0);
    return 0;
}
