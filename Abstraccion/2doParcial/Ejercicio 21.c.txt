#include <stdio.h>
#include <stdlib.h>

typedef struct Coche {
	
	char marca[30];
	struct Coche* siguiente; //aqui se encuentra el puntero al mismo tipo
} Coche;


int main(){
	Coche c1 = {"Toyota", NULL};
	Coche c2 = {"Volkswagen", NULL};
	c1.siguiente = &c2;
	
	Coche* ptr = &c1;
	while(ptr != NULL){
		
		printf("Coche: %s\n",ptr->marca);
		
	}
	return 0;
	
}
